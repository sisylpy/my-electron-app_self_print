## 💻 完整代码示例

### **Java完整示例：**

```java
import java.io.*;
import java.net.*;
import java.nio.charset.StandardCharsets;
import java.util.Properties;

public class SimplePrintRecorder {
    
    private String apiBaseUrl;
    private int marketId;
    private int deviceId;
    
    // 构造函数：从配置文件加载
    public SimplePrintRecorder(String configFile) throws IOException {
        Properties props = new Properties();
        props.load(new FileInputStream(configFile));
        
        this.apiBaseUrl = props.getProperty("api.base.url");
        this.marketId = Integer.parseInt(props.getProperty("market.id"));
        this.deviceId = Integer.parseInt(props.getProperty("device.id"));
    }
    
    // 打印并记录
    public void printAndRecord(int billId, int distributerId, 
                              String content, double billTotal, String billTradeNo) {
        try {
            // 1. 执行打印
            System.out.println("正在打印...");
            // TODO: 您的打印代码
            
            // 2. 记录打印
            String url = apiBaseUrl + "/api/machine/print/record";
            String params = String.format(
                "billId=%d&deviceId=%d&marketId=%d&distributerId=%d&billTotal=%.2f&billTradeNo=%s",
                billId, deviceId, marketId, distributerId, billTotal,
                URLEncoder.encode(billTradeNo, "UTF-8")
            );
            
            HttpURLConnection conn = (HttpURLConnection) new URL(url).openConnection();
            conn.setRequestMethod("POST");
            conn.setRequestProperty("Content-Type", "application/x-www-form-urlencoded");
            conn.setDoOutput(true);
            
            conn.getOutputStream().write(params.getBytes(StandardCharsets.UTF_8));
            
            int code = conn.getResponseCode();
            if (code == 200) {
                System.out.println("✅ 打印记录已保存");
            } else {
                System.err.println("❌ 记录失败: HTTP " + code);
            }
            
        } catch (Exception e) {
            System.err.println("打印失败: " + e.getMessage());
        }
    }
    
    public static void main(String[] args) throws IOException {
        // 使用示例
        SimplePrintRecorder recorder = new SimplePrintRecorder("printer.properties");
        
        recorder.printAndRecord(
            1001,                    // billId
            5,                       // distributerId
            "打印内容...",            // content
            158.50,                  // billTotal
            "BL202410150001"         // billTradeNo
        );
    }
}
```

**配置文件（printer.properties）：**
```properties
api.base.url=http://yourdomain.com/nongxinle
market.id=1
device.id=1
```

## ⚡ 最简实现（仅50行代码）

```java
// 最简单的实现，直接在您的打印方法中添加：

public void printBill(int billId, int distributerId, 
                     double billTotal, String billTradeNo) {
    // 您的打印代码...
    
    // 添加这一行即可！⭐
    new Thread(() -> {
        try {
            String url = "http://yourdomain.com/nongxinle/api/machine/print/record";
            String params = String.format(
                "billId=%d&deviceId=1&marketId=1&distributerId=%d&billTotal=%.2f&billTradeNo=%s",
                billId, distributerId, billTotal, billTradeNo
            );
            
            HttpURLConnection conn = (HttpURLConnection) new URL(url).openConnection();
            conn.setRequestMethod("POST");
            conn.setRequestProperty("Content-Type", "application/x-www-form-urlencoded");
            conn.setDoOutput(true);
            conn.getOutputStream().write(params.getBytes("UTF-8"));
            
            if (conn.getResponseCode() == 200) {
                System.out.println("记录成功");
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }).start();
}
```
