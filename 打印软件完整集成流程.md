## 💻 核心代码

### **最简实现（100行代码）：**

```java
public class SimplePrintSoftware {
    
    private String apiUrl = "http://yourdomain.com/nongxinle";
    private int marketId;
    private int deviceId;
    private int managerId;
    
    /**
     * 1. 登录
     */
    public boolean login(String phone) throws Exception {
        String url = apiUrl + "/api/machine/printSoftware/login";
        String params = "phone=" + phone;
        
        HttpURLConnection conn = (HttpURLConnection) new URL(url).openConnection();
        conn.setRequestMethod("POST");
        conn.setRequestProperty("Content-Type", "application/x-www-form-urlencoded");
        conn.setDoOutput(true);
        conn.getOutputStream().write(params.getBytes("UTF-8"));
        
        if (conn.getResponseCode() == 200) {
            // 解析响应
            JSONObject json = parseResponse(conn);
            
            if (json.getInt("code") == 0) {
                // 保存信息
                this.managerId = json.getInt("managerId");
                this.marketId = json.getInt("marketId");
                
                System.out.println("✅ 登录成功！市场：" + json.getString("marketName"));
                return true;
            }
        }
        return false;
    }
    
    /**
     * 2. 获取设备列表并选择
     */
    public void selectDevice() throws Exception {
        String url = apiUrl + "/api/machine/printSoftware/devices?marketId=" + marketId;
        
        HttpURLConnection conn = (HttpURLConnection) new URL(url).openConnection();
        conn.setRequestMethod("GET");
        
        if (conn.getResponseCode() == 200) {
            JSONObject json = parseResponse(conn);
            
            if (json.getInt("code") == 0) {
                JSONArray devices = json.getJSONArray("list");
                
                // TODO: 显示设备列表让用户选择
                // 这里假设选择了第一个设备
                JSONObject firstDevice = devices.getJSONObject(0);
                this.deviceId = firstDevice.getInt("nxPdId");
                
                System.out.println("✅ 已选择设备：" + firstDevice.getString("nxPdDeviceName"));
                
                // 保存配置
                saveConfig();
            }
        }
    }
    
    /**
     * 3. 打印并记录
     */
    public void printBill(int billId, int distributerId, String content,
                         double billTotal, String billTradeNo) throws Exception {
        // 3.1 执行打印
        System.out.println("正在打印...");
        // TODO: 您的打印代码
        
        // 3.2 记录打印
        String url = apiUrl + "/api/machine/print/record";
        String params = String.format(
            "billId=%d&deviceId=%d&marketId=%d&distributerId=%d&billTotal=%.2f&billTradeNo=%s",
            billId, deviceId, marketId, distributerId, billTotal, billTradeNo
        );
        
        HttpURLConnection conn = (HttpURLConnection) new URL(url).openConnection();
        conn.setRequestMethod("POST");
        conn.setRequestProperty("Content-Type", "application/x-www-form-urlencoded");
        conn.setDoOutput(true);
        conn.getOutputStream().write(params.getBytes("UTF-8"));
        
        if (conn.getResponseCode() == 200) {
            JSONObject json = parseResponse(conn);
            if (json.getInt("code") == 0) {
                System.out.println("✅ 打印记录已保存");
            }
        }
    }
    
    /**
     * 保存配置
     */
    private void saveConfig() throws IOException {
        Properties props = new Properties();
        props.setProperty("manager.id", String.valueOf(managerId));
        props.setProperty("market.id", String.valueOf(marketId));
        props.setProperty("device.id", String.valueOf(deviceId));
        props.store(new FileOutputStream("printer_config.properties"), null);
    }
    
    /**
     * 解析响应
     */
    private JSONObject parseResponse(HttpURLConnection conn) throws Exception {
        BufferedReader br = new BufferedReader(
            new InputStreamReader(conn.getInputStream(), "UTF-8")
        );
        StringBuilder sb = new StringBuilder();
        String line;
        while ((line = br.readLine()) != null) {
            sb.append(line);
        }
        br.close();
        return new JSONObject(sb.toString());
    }
}
```
